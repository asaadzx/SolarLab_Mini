```python
"""One-shot HTTP server for the SolarLab Mini JSON API."""

import socket
import json


class WebServer:
    """Serve the latest sensor reading to at most one client per call."""

    def __init__(self, get_data_callback, port=80):
        self.get_data = get_data_callback
        self.s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        self.s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        self.s.bind(('0.0.0.0', port))
        self.s.listen(5)
        self.s.settimeout(1)

    def handle_once(self):
        """Accept one client, reply with JSON, and never block the sensor loop."""
        try:
            conn, addr = self.s.accept()
        except OSError:
            return
        try:
            conn.settimeout(2)
            conn.recv(1024)
            data = self.get_data()
            body = json.dumps(data)
            response = (
                'HTTP/1.1 200 OK\r\n'
                'Content-Type: application/json\r\n'
                'Access-Control-Allow-Origin: *\r\n'
                'Content-Length: ' + str(len(body)) + '\r\n'
                'Connection: close\r\n'
                '\r\n' + body
            )
            conn.sendall(response.encode())
        except OSError as e:
            print('server error', e)
        finally:
            conn.close()
```

```python
import json
import sys
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

import pytest

pytest.importorskip("flask")

sys.path.insert(0, str(Path(__file__).resolve().parent.parent / "laptop_app" / "dashboard"))

import app as dashboard  # noqa: E402


class StubHandler(BaseHTTPRequestHandler):
    PAYLOAD = {"voltage": 4.536, "current": 0.125, "power": 0.567, "timestamp": 1234}

    def do_GET(self):
        body = json.dumps(self.PAYLOAD).encode()
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, *args):
        pass


@pytest.fixture
def stub_server(monkeypatch):
    server = ThreadingHTTPServer(("127.0.0.1", 0), StubHandler)
    thread = threading.Thread(target=server.serve_forever, daemon=True)
    thread.start()
    host, port = server.server_address[:2]
    monkeypatch.setattr(dashboard, "ESP_URL", "http://{}:{}/data".format(host, port))
    yield
    server.shutdown()
    server.server_close()
    thread.join(timeout=2)


def test_index_renders(stub_server):
    response = dashboard.app.test_client().get("/")
    assert response.status_code == 200
    assert b"SolarLab Mini" in response.data


def test_api_latest_proxies_json(stub_server):
    response = dashboard.app.test_client().get("/api/latest")
    assert response.status_code == 200
    assert response.get_json() == StubHandler.PAYLOAD


def test_api_latest_reports_connection_error(monkeypatch):
    monkeypatch.setattr(dashboard, "ESP_URL", "http://127.0.0.1:1/data")
    response = dashboard.app.test_client().get("/api/latest")
    assert response.status_code == 200
    assert "error" in response.get_json()
```
